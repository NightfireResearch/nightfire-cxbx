# (Included by the other scripts' first lines: from boot to the main menu, timed by page.)
waitpage 0x40000009 3000   # P_START
wait 500                   # until "Press START" shows
press START
waitpage 0x40000043        # P_ESTHERO, the movie before the main menu
wait 150
press A                    # skip it
waitpage 0x40000002        # P_MAIN
wait 60
# The join page's per-agent steps (C_RBMPSETUP) in a team game: team, character, handicap, back twice, then
# forward to ready, which goes on to the options page.
press DOWN
wait 30
press A                    # Multiplayer
waitpage 0x40000019        # P_MPJOIN
wait 100
press A
wait 120
press A
wait 120
press A
waitpage 0x4000001a        # P_MPSCENARIO, starting on Arena
wait 100
press DOWN                 # Team Arena
wait 40
press A
waitpage 0x40000013        # P_MPMAP
wait 100
press A
waitpage 0x40000051        # P_MPSETUP
wait 100
shot team
press RIGHT
wait 40
shot team_right
press A
wait 60
shot character
press RIGHT
wait 40
shot character_right
press A
wait 60
shot handicap
press RIGHT
wait 40
shot handicap_right
press B
wait 60
shot back_to_character
press B
wait 60
shot back_to_team
press A
wait 60
press A
wait 60
press A
waitpage 0x40000012 300    # P_MPOPTIONS once every joined player is ready
wait 100
shot options
quit
