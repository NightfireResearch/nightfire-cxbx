# A Quick Game played as blueprint (the mode poked before the match starts), for the MP and bot ports.
# (Included by the other scripts' first lines: from boot to the main menu, timed by page.)
waitpage 0x40000009 3000   # P_START
wait 500                   # until "Press START" shows
press START
waitpage 0x40000043        # P_ESTHERO, the movie before the main menu
wait 150
press A                    # skip it
waitpage 0x40000002        # P_MAIN
wait 60
# Quick Game from the scenario wheel.
press DOWN
wait 30
press A                    # Multiplayer
waitpage 0x40000019        # P_MPJOIN
wait 100
press A
wait 120
press A
wait 120
press A
waitpage 0x4000001a        # P_MPSCENARIO: the wheel starts on the last mode played (Arena), Quick Game is above it
wait 100
press UP
wait 40
shot quickgame_item
# !!! WARNING: seeding fixes the LEVEL only. The match after it is NOT repeatable: game logic runs on real
# !!! elapsed time, so bots, spawns and physics drift between runs. Do not compare the 'match' shot pixel for pixel.
seed                       # the Quick Game's level is random: the same one on every run
press A
waitpage 0x40000049 400    # P_MPCONFIRM
wait 100
poke 0x0026003c 0x20000100    # MPSettings.GameMode: blueprint
press A                    # start the match
wait 1500
shot match_1500
wait 1500
shot match_3000
press START                # the pause menu's scores
wait 120
shot pause
press RIGHT
wait 60
shot pause_tab2
press RIGHT
wait 60
shot pause_tab3
quit
